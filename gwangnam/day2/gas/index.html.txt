<script>
const $ = s => document.querySelector(s);
const PASTEL = ['#FFE9EE', '#FFF1DC', '#E6F7EE', '#E6F0FF', '#F1E9FF'];
const AL = {1:'난류',2:'우유',3:'메밀',4:'땅콩',5:'대두',6:'밀',7:'고등어',8:'게',9:'새우',10:'돼지고기',11:'복숭아',12:'토마토',13:'아황산류',14:'호두',15:'닭고기',16:'쇠고기',17:'오징어',18:'조개류',19:'잣'};
const esc = t => String(t).replace(/[&<>"]/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c]));
let mealOffset = 0, info = {};

// 탭
document.querySelectorAll('.tab').forEach(b => b.onclick = () => {
  document.querySelectorAll('.tab').forEach(x => x.setAttribute('aria-selected', x === b));
  document.querySelectorAll('.panel').forEach(p => p.classList.toggle('on', p.id === b.dataset.p));
  if (b.dataset.p === 'meal' && !$('#mbody').dataset.loaded) loadMeal(0);
});

// 시작
google.script.run.withSuccessHandler(i => {
  info = i;
  $('#title').textContent = i.school + ' ' + i.className + ' 도우미';
  document.title = $('#title').textContent;
  if (i.character) $('#ch').src = i.character;
}).getInfo();

// ④ 칭찬
function drawPraise(r) {
  $('#cnt').textContent = r.count ? r.count + '개!' : '';
  $('#plist').innerHTML = r.items.length ? r.items.map((p, k) =>
    `<div class="p" style="background:${PASTEL[k % PASTEL.length]}"><button class="x" title="지우기(선생님)" onclick="del(${p.row}, this)">×</button><b>${esc(p.to)}에게 💛</b><p>${esc(p.msg)}</p><small>— ${esc(p.from)} · ${esc(p.time)}</small></div>`).join('')
    : '<div class="msg">첫 번째 칭찬을 남겨 주세요!</div>';
}
function loadPraise() { google.script.run.withSuccessHandler(drawPraise).withFailureHandler(e => $('#plist').innerHTML = '<div class="msg">불러오지 못했어요: ' + esc(e.message) + '</div>').listPraise(); }
$('#send').onclick = () => {
  const to = $('#to').value.trim(), text = $('#text').value.trim(), from = $('#from').value.trim();
  if (!to || !text || !from) { $('#pmsg').textContent = '빈 칸을 모두 채워 주세요 🙂'; return; }
  $('#send').disabled = true; $('#pmsg').textContent = '보내는 중…';
  google.script.run.withSuccessHandler(r => {
    $('#send').disabled = false; $('#pmsg').textContent = r.message;
    if (r.ok) { $('#to').value = ''; $('#text').value = ''; drawPraise(r.list); setTimeout(() => $('#pmsg').textContent = '', 2500); }
  }).withFailureHandler(e => { $('#send').disabled = false; $('#pmsg').textContent = '저장하지 못했어요: ' + e.message; }).savePraise(to, text, from);
};
window.del = (row, btn) => {
  const card = btn.parentElement;
  if (card.querySelector('.pin')) return;
  const f = document.createElement('div'); f.className = 'pin';
  f.innerHTML = '<input type="password" placeholder="선생님 비밀번호" style="padding:8px;font-size:15px"><button class="go" style="margin-top:6px;padding:8px;font-size:16px">지우기</button>';
  card.appendChild(f);
  f.querySelector('button').onclick = () => google.script.run.withSuccessHandler(r => { if (r.ok) drawPraise(r.list); else $('#pmsg').textContent = r.message; }).deletePraise(row, f.querySelector('input').value);
};
loadPraise();

// ⑤ 급식
const mealMemo = {};
function drawMeal(r) {
  $('#mbody').dataset.loaded = 1;
  $('#mdate').textContent = (r.school ? r.school + ' · ' : '') + (r.date || '');
  if (!r.ok || r.empty) { $('#mbody').innerHTML = `<div class="msg big">${esc(r.message)}</div>`; $('#allergyKey').textContent = ''; return; }
  $('#mbody').innerHTML = r.meals.map(m => `<span class="tag">${esc(m.type)} · ${esc(m.cal)}</span>` + m.dishes.map(d =>
    `<div class="dish"><b>${esc(d.name)}</b><span>${d.allergy.map(n => AL[n]).join(' · ')}</span></div>`).join('')).join('<br>');
  $('#allergyKey').textContent = '알레르기 번호: ' + Object.entries(AL).map(([k, v]) => k + ' ' + v).join(', ');
}
function loadMeal(o) {
  mealOffset = o;
  document.querySelectorAll('.nav button').forEach(b => b.classList.toggle('on', Number(b.dataset.o) === o));
  $('#aiout').innerHTML = ''; $('#aimsg').textContent = '';
  if (mealMemo[o]) return drawMeal(mealMemo[o]);
  $('#mdate').textContent = '급식을 가져오는 중…'; $('#mbody').innerHTML = '';
  google.script.run.withSuccessHandler(r => { mealMemo[o] = r; drawMeal(r); $('#aidate').textContent = (r.date || '') + ' 급식을 AI가 읽고 알레르기와 영양을 알려 줘요.'; })
    .withFailureHandler(e => drawMeal({ ok: false, message: '급식을 가져오지 못했어요: ' + e.message })).getMeal(o);
}
document.querySelectorAll('.nav button').forEach(b => b.onclick = () => loadMeal(Number(b.dataset.o)));

// ⑥ AI 영양사
$('#ask').onclick = () => {
  $('#ask').disabled = true; $('#aimsg').textContent = 'AI가 급식을 살펴보는 중… 🍳'; $('#aiout').innerHTML = '';
  google.script.run.withSuccessHandler(r => {
    $('#ask').disabled = false; $('#aimsg').textContent = r.ok ? (r.saved ? '저장된 분석이에요 (같은 날은 다시 부르지 않아요)' : '') : r.message;
    if (r.ok) $('#aiout').innerHTML = `<div class="ai-out">${esc(r.text)}</div><div class="note">AI가 만든 안내예요. 정확한 내용은 영양 선생님께 확인하세요.</div>`;
  }).withFailureHandler(e => { $('#ask').disabled = false; $('#aimsg').textContent = '분석하지 못했어요: ' + e.message; }).analyzeMeal(mealOffset);
};
</script>
